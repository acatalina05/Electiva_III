Detección de outliers

def detect_outliers_iqr(x):
    Q1 = x.quantile(0.25)
    Q3 = x.quantile(0.75)
    IQR = Q3 - Q1
    lower = Q1 - 1.5 * IQR
    upper = Q3 + 1.5 * IQR
    return (x < lower) | (x > upper)

df['outlier_iqr'] = df.groupby('CountryCode')['Value'].transform(detect_outliers_iqr)
